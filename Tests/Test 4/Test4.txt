; ------------------------------------------------------------
; Test4.txt : Floating-Point Accumulator
; ------------------------------------------------------------
; WHAT THIS PROGRAM DOES (load it at address 00):
;  Demonstrates the floating-point ADD instruction (opcode 6).
;  Vole floats are 8 bits: 1 sign bit, 3 exponent bits (excess 4)
;  and a 4-bit mantissa (value = 0.mmmm * 2^(exp-4)).
;   Part 1: A loop adds +0.5 (pattern 48) to a running total
;           14 times and prints the pattern after each addition.
;           The total climbs 0.5, 1.0, 1.5, ... up to 7.0.
;   Part 2: Adds -0.5 (pattern C8) three times, printing each
;           result (7.0 -> 6.5 -> 6.0 -> 5.5).
;   Part 3: Adds 1.5 (pattern 5C) and 2.25 (pattern 69) and
;           prints 3.75 (pattern 6F).
;  The values were chosen so every result is exactly
;  representable and no rounding occurs.
;
;  Register usage:
;   R0 = 0 (zero register)  R1 = 48 (+0.5)  R2 = running total
;   R3 = loop counter (14)  R4 = FF (-1)    R5 = C8 (-0.5)
;   R6 = 5C (1.5)  R7 = 69 (2.25)  R8 = R6 + R7 (float)
; ------------------------------------------------------------

2000   ; R0 = 0
2148   ; R1 = 48 (+0.5)
2200   ; R2 = 00 (0.0)
230E   ; R3 = 14
24FF   ; R4 = FF (-1)
6221   ; LOOP: R2 = R2 + R1 (float)
32FF   ; print R2
5334   ; R3 = R3 - 1
B314   ; if R3 == 0 jump to 14 (part 2)
B00A   ; jump back to LOOP (0A)
25C8   ; R5 = C8 (-0.5)
6225   ; R2 = R2 + R5 (float)
32FF   ; print R2
6225   ; R2 = R2 + R5
32FF   ; print R2
6225   ; R2 = R2 + R5
32FF   ; print R2
265C   ; R6 = 5C (1.5)
2769   ; R7 = 69 (2.25)
6867   ; R8 = R6 + R7 (float)
38FF   ; print R8
C000   ; halt

; ------------------------------------------------------------
; EXPECTED OUTPUT (screen, in order)
;  Decimal: 72 88 92 104 106 108 110 120 121 122 123 124 125 126
;           125 124 123 111
;  Hex:     48 58 5C 68 6A 6C 6E 78 79 7A 7B 7C 7D 7E
;           7D 7C 7B 6F
;  As floating-point values:
;           0.5 1.0 1.5 2.0 2.5 3.0 3.5 4.0 4.5 5.0 5.5 6.0 6.5 7.0
;           6.5 6.0 5.5 3.75
;  (The screen shows the raw 8-bit patterns, not the decoded floats.)
; ------------------------------------------------------------
