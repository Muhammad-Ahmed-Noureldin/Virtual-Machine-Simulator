; ------------------------------------------------------------
; Test2.txt : Factorial Table (1! to 5!) using nested loops
; ------------------------------------------------------------
; WHAT THIS PROGRAM DOES (load it at address 00):
;  The Vole machine has no multiply instruction, so this program
;  multiplies with repeated addition. An outer loop walks i from
;  2 to 5. For every i, an inner loop adds the current result to
;  an accumulator i times (result * i). The accumulator then
;  becomes the new result, which is printed. Starting from
;  1! = 1 this prints 1!, 2!, 3!, 4! and 5!.
;
;  Register usage:
;   R0 = 0 (zero register)        R1 = running factorial result
;   R2 = i (multiplier, 2..5)     R3 = inner loop counter
;   R4 = accumulator for result*i R5 = FF (-1)
;   R6 = 1                        R7 = outer loop counter (4)
; ------------------------------------------------------------

2000   ; R0 = 0
2101   ; R1 = 1  (1!)
2202   ; R2 = 2  (first multiplier)
2704   ; R7 = 4  (outer iterations)
25FF   ; R5 = FF (-1)
2601   ; R6 = 1
31FF   ; print 1! 
4023   ; OUTER: R3 = R2 (inner counter = i)
2400   ; R4 = 0
5441   ; INNER: R4 = R4 + R1
5335   ; R3 = R3 - 1
B31A   ; if R3 == 0 leave inner loop (1A)
B012   ; jump back to INNER (12)
4041   ; R1 = R4 (new factorial)
31FF   ; print R1
5226   ; R2 = R2 + 1
5775   ; R7 = R7 - 1
B726   ; if R7 == 0 jump to halt (26)
B00E   ; jump back to OUTER (0E)
C000   ; halt

;  Hex:     01 02 06 18 78
;  Meaning: 1!, 2!, 3!, 4!, 5!
;  Final registers: R1 = 78 (120), R2 = 06, R7 = 00
